\begin{frame}{The KV cache formula}
  \llmdiagKV
  \begin{equation*}
    \underbrace{\text{KV bytes per token}}_{\text{per user, per token}}
    \;=\;
    \underbrace{2}_{\mathclap{\text{K and V}}}
    \;\times\; \text{layers}
    \;\times\; \text{kv\_heads}
    \;\times\; \text{head\_dim}
    \;\times\; \text{bytes per value}
  \end{equation*}
  \vspace{1.2em}
  \begin{itemize}
    \item Each token that you generated stays in memory.
    \item It grows \emph{linearly} with conversation length.
    \item You pay it \emph{once per concurrent user}.
  \end{itemize}
  \vspace{0.6em}
  \emphc{layers, kv\_heads, head\_dim, bytes per value --- all in \texttt{config.json}.}
\end{frame}
\note{\textbf{5:30.} The one fact about attention I assume: decoding needs the keys
and values of every previous token, so you keep them.}

% -------------------------------------------------------------------------
\begin{frame}{Llama-3-8B: 128\,KiB per token, 1\,GiB per conversation}
  \llmdiagKV
  \textbf{Llama-3-8B:} 32 layers, 8 KV heads, head\_dim 128, fp16 cache.
  \vspace{0.6em}
  \begin{align*}
    \text{KV per token}
      &= 2 \times 32 \times 8 \times 128 \times 2\ \text{bytes} \\[0.2em]
    \uncover<2->{&= 131{,}072\ \text{bytes} \;=\; \emphc{128\ \text{KiB per token}}} \\[0.6em]
    \uncover<3->{\text{An 8k conversation} &= 8192 \times 128\ \text{KiB}
      \;=\; \warnc{1\ \text{GiB} \ \text{for one user}}}
  \end{align*}
  \vspace{0.4em}
  \uncover<3->{You pay 16\,GB for the weights once. You pay this 1\,GiB again for
  each person who opens a chat window.}
\end{frame}

  \end{center}
  \vspace{0.4em}
  \begin{itemize}\small
    \item In \textbf{MHA}, each query head has its own key-value head.
          Cache size scales with the number of heads.
    \item In \textbf{GQA}, $G$ query heads share one KV head.
          Llama-3-8B uses $G{=}4$: 32 query heads, 8 KV heads.
    \item In \textbf{MQA}, all query heads share one KV head.
          MQA is faster than GQA, but quality degrades on long output.
  \end{itemize}
  \vspace{0.3em}

    Ainslie et al.\ uptrained T5-XXL from MHA to GQA-8 in 5\% of the original
    training steps. Inference was \emphc{5.4$\times$ faster} (1.51\,s
    $\rightarrow$ 0.28\,s) with no measurable quality loss (ROUGE-1: 46.2
    $\rightarrow$ 46.8).
  \end{center}
\end{frame}

path converts an MHA checkpoint to GQA by mean-pooling the KV projection
matrices, then training for 5\% more steps.

\section{Why decoding is slow}
% =========================================================================

\begin{frame}{Prefill is compute-bound. Decode is bandwidth-bound.}
  \llmdiagK
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Prefill} (your prompt)\\[0.3em]
      \small All prompt tokens go through the network \emph{in parallel}.
      Big matrix multiplies keep the arithmetic units of the GPU busy.
      \emph{Compute-bound.}

      \small One token at a time, in sequence. For each token, the GPU reads
      \emph{every weight} out of HBM. The GPU spends its time on byte
      traffic. \emph{Bandwidth-bound.}
    \end{column}
  \end{columns}

      \end{block}
    \end{minipage}
  \end{center}

\begin{frame}{AWQ, GPTQ, NVFP4: three ways to shrink the weights}
  \llmdiagW
  \vspace{-0.2em}
  \small
  \begin{itemize}\setlength{\itemsep}{0.6em}
    \item \textbf{GPTQ} (Frantar et al., ICLR 2023) quantizes weights one
          column at a time using an approximate second-order method. Each
          column is rounded, and the rounding error is compensated by
          adjusting all remaining columns via the inverse Hessian. A
          calibration set of $\sim$128 examples computes the Hessian.
          GPTQ runs in minutes on one GPU.
    \item \textbf{AWQ} (Lin et al., MLSys 2024) observes that $\sim$1\% of
          weights process large-magnitude activations and matter
          disproportionately. AWQ applies per-channel scaling before
          quantization: it multiplies important weight channels by a scale
          factor and divides the activations by the same factor, shifting
          the quantization grid to protect those weights.
    \item \textbf{NVFP4} stores each weight in 4 bits of floating-point
          (1 sign, 2 exponent, 1 mantissa). A block of 16 values shares
          an E4M3 scale factor, and the tensor has a global FP32 scale.
          Requires Blackwell fifth-generation Tensor Cores.
  \end{itemize}
\end{frame}
\note{This slide is for the audience member who asks ``what does AWQ actually
do?'' The key difference: GPTQ compensates rounding error through the Hessian.
AWQ protects salient weights by rescaling before quantization. NVFP4 uses a
hardware-native 4-bit float format with fine-grained block scaling.

arXiv:2210.17323 (GPTQ), arXiv:2306.00978 (AWQ).}

    % Fused: 1 round-trip
    \node[font=\tiny\bfseries, black!60] at (-1.8, -0.1) {fused};

    \node[op, draw=good, fill=good!10, minimum width=4.6cm] (fused)
      at (1.5, -0.1) {RMSNorm + RoPE + SwiGLU + Quant};
    \draw[ar, good] (fused.north) -- ++(0,0.25);
    \draw[ar, good] (fused.north) ++(0.15,0.25) -- ++(0,-0.25);
    \node[font=\tiny, good!60, align=center] at (5.2, 0.15) {$1\times$\\read+write};
    \end{tikzpicture}
  \end{center}
  \vspace{0.2em}
  \scriptsize
  \begin{center}
    \begin{tabular}{@{}llp{6.5cm}@{}}
      \toprule
      & \textbf{approach} & \textbf{how it fuses} \\
      \midrule
      \textbf{FlashAttention} & hand-written CUDA
        & fuses Q$\cdot$K$^T$ $\rightarrow$ softmax $\rightarrow$ $\times$V
          in SRAM tiles, avoids materializing the $O(N^2)$ attention matrix \\
      \textbf{Triton} & Python DSL
        & tile-level GPU code, compiler handles memory staging \\
      \textbf{torch.compile} & automatic
        & traces PyTorch, applies fusion passes, emits Triton kernels \\
      \textbf{JAX XLA} & automatic
        & HLO fusion passes over the computation graph \\
      \textbf{TensorRT} & offline
        & graph-level layer fusion + per-GPU GEMM auto-tuning \\
      \textbf{FlashRT} & hand-written + CUDA graph
        & fused memory-bound ops, delegates GEMM to cuBLAS \\
      \bottomrule
    \end{tabular}
  \end{center}
\end{frame}
\note{The single principle: a chain of N unfused kernels does N HBM
round-trips. A fused kernel reads HBM once and writes once, cutting memory
traffic by up to N$\times$. The spectrum runs from fully manual (FlashAttention,
FlashRT) to fully automatic (torch.compile, XLA), with Triton and TensorRT in
between. Manual fusion achieves the highest efficiency but requires GPU
programming expertise.

Liger-Kernel applies the same technique to training (fused RMSNorm + SwiGLU),
claiming +20\% throughput and $-$60\% memory.}

    \end{tikzpicture}
  \end{center}
  \vspace{0.2em}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \small
      \textbf{Why it works:} decode is bandwidth-bound, so the GPU has spare
      compute. Verifying $k$ draft tokens reads the weights \textbf{once}
      (same bandwidth as 1 token) but uses $k\times$ the arithmetic.
    \end{column}
    \begin{column}{0.44\textwidth}
      \small
      \textbf{Why n-gram drafts work:} the acceptance rate only needs to
      exceed $1/k$ to break even. Natural language has enough local
      repetition (articles, prepositions, closing brackets) that even a
      simple n-gram model achieves 50--70\% acceptance.
    \end{column}
  \end{columns}
  \vspace{0.3em}
  \begin{center}\small
    Modified rejection sampling preserves the \emphc{exact target distribution}.
  \end{center}
\end{frame}
\note{The diagram shows 5 draft tokens verified in one forward pass. Three
accepted (green), one rejected (orange ``on'' replaced by ``upon''), and the
correction token sampled from the residual distribution. The key insight: the
big model reads its weights once for all 5 tokens, so the bandwidth cost is
1 token while producing up to 5. An n-gram draft model works because even
50\% acceptance at $k=5$ gives 2.5 tokens per weight read.

The identical-distribution guarantee comes from modified rejection sampling:
accept if $p_\text{target} \geq p_\text{draft}$, otherwise accept with
probability $p_\text{target}/p_\text{draft}$ and sample a correction from
the residual. Leviathan et al., arXiv:2302.01318.}

\begin{frame}{Multi-Token Prediction trains the draft head into the model}
  \textbf{MTP} trains a model to predict $D$ tokens ahead. At inference
  the extra heads serve as a \emph{built-in draft} for speculative decoding,
  so no separate draft model is needed.
  \vspace{0.6em}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{Training}\\[0.3em]
      \small
      $D$ auxiliary heads share the backbone and the embedding layer. Each head
      predicts the token $d$ positions ahead. The MTP loss is averaged and
      weighted by $\lambda$ (DeepSeek-V3 uses $\lambda{=}0.3$ for the first 10T
      tokens, then $0.1$).
    \end{column}
    \begin{column}{0.44\textwidth}
      \textbf{Inference}\\[0.3em]
      \small
      The auxiliary heads propose tokens, and the main model verifies them in
      one forward pass (speculative decoding). The heads can also be
      discarded, and the main model runs independently.
    \end{column}
  \end{columns}
  \vspace{0.8em}
  \begin{center}\small
    DeepSeek-V3 ablation ($D{=}1$): HumanEval improved from 20.7\% to 26.8\%
    (+6.1) on the small MoE, and from 44.5\% to 53.7\% (+9.2) on the large
    MoE. The auxiliary loss regularizes training, so quality improves even
    when the extra heads are discarded at inference.
  \end{center}
\end{frame}
\note{The training effect is larger than the inference effect: the auxiliary
heads regularize and improve quality even when discarded. The inference
bonus is speculative decoding with no separate draft model.

DeepSeek-V3 technical report (arXiv:2412.19437). NVIDIA NeMo supports MTP via
\texttt{mtp\_num\_layers} and \texttt{mtp\_loss\_scaling\_factor}.}

\begin{frame}{vLLM-Omni: a pipeline of stages, and one stage emits actions}

\begin{frame}{FlashRT: no compile, hand-written kernels, one CUDA graph}
  \llmdiagK
  The opposite design point from TensorRT-LLM:
  \emphc{you run it without a compile step or an ONNX export}.
  \vspace{0.6em}
  \begin{itemize}
    \item \textbf{Hand-written CUDA kernels} for the memory-bound operations:
          norm, activation, RoPE, quantization.
    \item \textbf{Delegates the compute-bound ones} to cuBLASLt / CUTLASS /
          FlashAttention-2.

    \item \textbf{Static FP8 (E4M3) and NVFP4}, with calibration scales cached
          on disk. The $(1+w)$ of RMSNorm folds into the next GEMM at load time.
  \end{itemize}
  \vspace{0.5em}
  \begin{center}\small
    Batch size 1 means dispatch overhead \emph{is} your latency budget.
  \end{center}
\end{frame}
\note{\textbf{30:30.} At batch 1 with a 20\,ms budget, Python dispatch of a
few hundred kernels takes a large fraction of the frame. One
CUDA graph replaces all of it with a single launch.}

\begin{frame}{How to benchmark and get a true number}
  \begin{enumerate}
    \item \textbf{Warm up first.} The first request pays for CUDA graph
          capture, allocator growth and JIT.
    \item \textbf{Fix the concurrency} and state it. A number without a batch
          size means nothing.
    \item \textbf{Use realistic input \emph{and} output lengths.}

          One of them alone misleads.
    \item \textbf{Re-run the arithmetic} from Parts 1--5 against what the
          server reports. The two must agree. If they do not, suspect the
          benchmark first.

\begin{frame}{Backup: NVFP4 stores weights in 4-bit floats on Blackwell}
  \small
  NVFP4 uses 1 sign bit, 2 exponent bits, and 1 mantissa bit. The
  representable range is $-6$ to $+6$.
  \vspace{0.4em}
  \begin{center}
    \begin{tabular}{lrrr}
      \toprule
      \textbf{format} & \textbf{bits} & \textbf{vs FP16} & \textbf{block scaling} \\
      \midrule
      FP8 (E4M3)  & 8 & 2$\times$ smaller  & per-tensor \\
      \emphc{NVFP4} & 4 & \emphc{3.5$\times$ smaller} & two-level: E4M3 per 16
        values, FP32 per tensor \\
      MXFP4       & 4 & 3.5$\times$ smaller & power-of-two per 32 values \\
      \bottomrule
    \end{tabular}
  \end{center}
  \vspace{0.4em}
  \begin{itemize}\small
    \item NVFP4 scales at \textbf{two levels}: each block of 16 values shares
          an E4M3 scale factor, and the whole tensor has an FP32 scale. The
          16-value block and the floating-point scale preserve more dynamic
          range than MXFP4, which uses 32-value blocks with a power-of-two
          scale.
    \item On DeepSeek-R1-0528, NVFP4 lost $\leq$1\% accuracy vs.\ FP8 across
          seven benchmarks. On AIME 2024 it was 2\% \emph{better} than FP8.
    \item Requires \textbf{Blackwell} (fifth-gen Tensor Cores). Not available
          on Hopper or older.
  \end{itemize}
\end{frame}
\note{Backup. MXFP4 uses a coarser block (32 values) with a power-of-two
scale, which loses more precision on outlier-heavy weight distributions.
NVFP4 keeps a floating-point scale on a 16-value block.

Source: NVIDIA developer blog, ``Introducing NVFP4''.}

      A distilled or smaller model that \emph{passes your eval} beats a
      big one that is too slow to use.\\[0.5em]
      Also: one base model with many \textbf{LoRA adapters} instead of many
      full fine-tunes. The adapters are megabytes. You load the base model
      once.\\[0.4em]
      \emph{(Adapters here are a serving trick. Training is another talk.)}
    \end{column}
    \begin{column}{0.49\textwidth}
      \textbf{\#7 More GPUs, last}\\[0.3em]
      \small
      \textbf{Tensor parallel} inside a node. It needs a fast interconnect
      (NVLink), or the bottleneck moves to the wire.\\[0.5em]
      \textbf{Pipeline parallel} across nodes.\\[0.5em]
      \textbf{CPU offload} as a last resort. It is slow for the same
      reason that decode is slow: PCIe bandwidth is more than 10$\times$
      less than HBM bandwidth, and decode is bandwidth-bound.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Backup: common questions}
  \small
  \begin{itemize}
    \item \textbf{Self-host or API?} API, unless sustained high volume, a
          fine-tuned model, or data that cannot leave the building. Do the
          \$/1M math.
    \item \textbf{Does quantization hurt quality?} Sometimes, and not
          uniformly across tasks. 4-bit weight-only is good on most tasks, but make sure
          that it passes \emph{your} eval, not perplexity.
    \item \textbf{Why is my model slow on a big GPU?} You are bandwidth-bound
          at batch 1. Batch it, or shrink the bytes.
    \item \textbf{vLLM, SGLang or TensorRT-LLM?} vLLM by default. SGLang for
          structured output and prefix reuse. TensorRT-LLM for the last 20\%
          on fixed NVIDIA hardware.
    \item \textbf{How much VRAM for a 70B?} int4 $\approx$ 35\,GB of weights,
          plus cache. In practice two 40\,GB cards or one 80\,GB card, and
          you will still want a quantized KV cache.
    \item \textbf{CPU offload?} You can, and it will be slow. PCIe is $>10\times$
          slower than HBM.
    \item \textbf{MoE?} Fewer \emph{active} parameters per token, so faster
          decode, but you still hold \emph{all} the experts in VRAM. Good
          for speed, no help for capacity.
  \end{itemize}
\end{frame}
